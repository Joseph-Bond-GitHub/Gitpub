
DROP TABLE IF EXISTS tickets;
DROP TABLE IF EXISTS screenings;
DROP TABLE IF EXISTS screens;
DROP TABLE IF EXISTS films;

CREATE TABLE films (
    film_id INTEGER PRIMARY KEY,
    title TEXT NOT NULL,
    genre TEXT NOT NULL,
    release_year INTEGER NOT NULL,
    duration_minutes INTEGER NOT NULL,
    age_rating TEXT NOT NULL
);

CREATE TABLE screens (
    screen_id INTEGER PRIMARY KEY,
    screen_name TEXT NOT NULL,
    capacity INTEGER NOT NULL,
    screen_type TEXT NOT NULL
);

CREATE TABLE screenings (
    screening_id INTEGER PRIMARY KEY,
    film_id INTEGER NOT NULL REFERENCES films(film_id),
    screen_id INTEGER NOT NULL REFERENCES screens(screen_id),
    screening_date DATE NOT NULL,
    start_time TIME NOT NULL,
    ticket_price NUMERIC(6,2) NOT NULL
);

CREATE TABLE tickets (
    ticket_id INTEGER PRIMARY KEY,
    screening_id INTEGER NOT NULL REFERENCES screenings(screening_id),
    customer_name TEXT NOT NULL,
    ticket_type TEXT NOT NULL,
    amount_paid NUMERIC(6,2) NOT NULL,
    status TEXT NOT NULL
);
